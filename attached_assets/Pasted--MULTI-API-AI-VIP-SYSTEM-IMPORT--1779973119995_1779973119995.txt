def get_data_binance(symbol):
try:
url = f"https://api.binance.com/api/v3/klines?symbol={symbol}&interval=5m&limit=100"
data = requests.get(url).json()
return data
except:
return None

def get_data_coingecko(symbol):
try:
url = f"https://api.coingecko.com/api/v3/coins/{symbol}/market_chart?vs_currency=usd&days=1"
data = requests.get(url).json()
return data
except:
return None

def get_data_cryptocompare(symbol):
try:
url = f"https://min-api.cryptocompare.com/data/v2/histominute?fsym={symbol}&tsym=USDT&limit=100"
data = requests.get(url).json()
return data
except:
return None
